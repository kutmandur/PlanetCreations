import React from 'react';
import { getGameColor } from '../../utils/helpers';

const Spinner = ({ gameId = 'default', size = 'large' }) => {
    const color = getGameColor(gameId);

    const sizeClasses = {
        small: 'w-5 h-5 border-2',
        medium: 'w-8 h-8 border-4',
        large: 'w-16 h-16 border-4',
    };
    
    // For the small spinner used in buttons, we render it directly without extra padding.
    if (size === 'small') {
        return (
            <div className={`${sizeClasses.small} ${color.border} border-t-transparent rounded-full animate-spin`}></div>
        );
    }
    
    // The original large spinner format with padding for page loads.
    return (
        <div className="flex justify-center items-center p-10">
            <div className={`${sizeClasses[size]} ${color.border} border-dashed rounded-full animate-spin`}></div>
        </div>
    );
};

export default Spinner;
